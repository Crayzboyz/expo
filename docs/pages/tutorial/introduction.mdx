// App.tsx
import React from "react";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";

export default function App() {
  const [step, setStep] = React.useState("welcome");
  const [phone, setPhone] = React.useState("");
  const [code, setCode] = React.useState("");
  const [pin, setPin] = React.useState("");
  const [profile, setProfile] = React.useState({ name: "", topicInterest: "" });

  if (step === "welcome") {
    return (
      <div className="flex flex-col items-center justify-center h-screen p-4">
        <h1 className="text-3xl font-bold mb-6">MÜNAZARA</h1>
        <Tabs defaultValue="giris" className="w-full max-w-sm">
          <TabsList>
            <TabsTrigger value="giris">Giriş Yap</TabsTrigger>
            <TabsTrigger value="kayit">Kayıt Ol</TabsTrigger>
          </TabsList>
          <TabsContent value="giris">
            <Input placeholder="PIN" type="password" className="mt-4" />
            <Button className="mt-4 w-full" onClick={() => setStep("profile")}>Giriş Yap</Button>
          </TabsContent>
          <TabsContent value="kayit">
            <Input placeholder="Telefon Numarası" value={phone} onChange={(e) => setPhone(e.target.value)} className="mt-4" />
            <Button className="mt-4 w-full" onClick={() => setStep("sms")}>SMS Gönder</Button>
          </TabsContent>
        </Tabs>
      </div>
    );
  }

  if (step === "sms") {
    return (
      <div className="flex flex-col items-center justify-center h-screen p-4">
        <h2 className="text-xl font-semibold mb-4">SMS Doğrulama</h2>
        <p className="text-sm mb-2">Telefon numarasına gelen kodu gir:</p>
        <Input placeholder="1234" value={code} onChange={(e) => setCode(e.target.value)} />
        <Button className="mt-4 w-full" onClick={() => setStep("pin")}>Doğrula</Button>
      </div>
    );
  }

  if (step === "pin") {
    return (
      <div className="flex flex-col items-center justify-center h-screen p-4">
        <h2 className="text-xl font-semibold mb-4">PIN Belirle</h2>
        <Input placeholder="PIN (4 haneli)" value={pin} onChange={(e) => setPin(e.target.value)} />
        <Button className="mt-4 w-full" onClick={() => setStep("profile")}>Devam</Button>
      </div>
    );
  }

  if (step === "profile") {
    return (
      <div className="flex flex-col items-center justify-center h-screen p-4">
        <h2 className="text-xl font-semibold mb-4">Profilini Oluştur</h2>
        <Input placeholder="Adınız" value={profile.name} onChange={(e) => setProfile({ ...profile, name: e.target.value })} />
        <Input placeholder="İlgi Duyduğun Konular" className="mt-2" value={profile.topicInterest} onChange={(e) => setProfile({ ...profile, topicInterest: e.target.value })} />
        <Button className="mt-4 w-full" onClick={() => setStep("home")}>Profil Tamamla</Button>
      </div>
    );
  }

  if (step === "home") {
    return (
      <div className="p-4">
        <h1 className="text-2xl font-bold mb-4">Hoşgeldin, {profile.name}</h1>
        <Card>
          <CardContent className="p-4">
            <h2 className="font-semibold">Münazara Odaları</h2>
            <p className="text-sm mt-2">Bir konuda merakını ifade et ve bir bilgi sahibiyle konuşmaya başla.</p>
            <Button className="mt-4" onClick={() => setStep("room")}>Oda Oluştur</Button>
          </CardContent>
        </Card>
      </div>
    );
  }

  if (step === "room") {
    return (
      <div className="p-4">
        <h1 className="text-xl font-bold mb-4">Münazara Odası</h1>
        <Card>
          <CardContent className="p-4">
            <div className="space-y-4">
              <div className="text-left">
                <span className="font-semibold">Sen:</span> Ben bu konuda şunu merak ediyorum...
              </div>
              <div className="text-right">
                <span className="font-semibold">Bilgi Sahibi:</span> Harika bir soru. Şöyle açıklayayım...
              </div>
            </div>
          </CardContent>
        </Card>
      </div>
    );
  }

  return null;
}
